/* a1_bonjour.c - TP2 exo A1
   codes de sortie :
   0 = ok
   2 = trop d'arguments
   3 = argument inconnu
*/
#include <stdio.h>
#include <string.h>

#define VERSION "1.0"

int main(int argc, char *argv[])
{
    /* plus d'un argument -> usage incorrect */
    if (argc > 2) {
        fprintf(stderr, "usage : %s [--version]\n", argv[0]);
        return 2;
    }

    if (argc == 2) {
        /* seul --version est accepte */
        if (strcmp(argv[1], "--version") == 0) {
            printf("version=%s\n", VERSION);
            return 0;
        }
        fprintf(stderr, "argument inconnu : %s\n", argv[1]);
        fprintf(stderr, "usage : %s [--version]\n", argv[0]);
        return 3;
    }

    /* pas d'argument : ligne d'identification */
    printf("outil=tp2-c version=%s compile=%s\n", VERSION, __DATE__);
    return 0;
}
